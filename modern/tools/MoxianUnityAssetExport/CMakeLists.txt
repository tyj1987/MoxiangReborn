cmake_minimum_required(VERSION 3.20)
project(MoxiangUnityAssetExport LANGUAGES CXX)
add_executable(mxh_unity_appearance_export appearance.cpp ../../src/character_appearance_catalog.cpp ../../src/mh_file_ex.cpp ../../src/game/item_list_parser.cpp)
target_include_directories(mxh_unity_appearance_export PRIVATE ../../include ../../third_party)
target_compile_features(mxh_unity_appearance_export PRIVATE cxx_std_20)
target_compile_definitions(mxh_unity_appearance_export PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(mxh_unity_appearance_export PRIVATE bcrypt)
if(MSVC)
  target_compile_options(mxh_unity_appearance_export PRIVATE /W4 /utf-8)
endif()

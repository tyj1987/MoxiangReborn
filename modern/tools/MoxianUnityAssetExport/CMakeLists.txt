cmake_minimum_required(VERSION 3.20)
project(MoxiangUnityAssetExport LANGUAGES CXX)
add_executable(mxh_unity_asset_export main.cpp ../../src/hfl_height_field.cpp)
target_include_directories(mxh_unity_asset_export PRIVATE ../../include)
target_compile_features(mxh_unity_asset_export PRIVATE cxx_std_20)
target_compile_definitions(mxh_unity_asset_export PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(mxh_unity_asset_export PRIVATE bcrypt)
if(MSVC)
  target_compile_options(mxh_unity_asset_export PRIVATE /W4 /utf-8)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME mxh_unity_asset_export_contract
  COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_export.py $<TARGET_FILE:mxh_unity_asset_export>)
